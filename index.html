Weave a graph.

Edit the JavaScript and watch the threads move. Try changing from 13 to 3.

String patterns
function pattern(ring) {
  const pins = ring(36)
  const skip = 13
  const color = '#ae4c2e'

  const links = pins.map((pin, i) => ({
    source: pin.id,
    target: pins[(i + skip) % pins.length].id
  }))

  return { nodes: pins, links, color }
}
